cmake_minimum_required(VERSION 3.16)
project(poker_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(poker_core STATIC cpp/engine.cpp cpp/fast_eval.cpp cpp/vendor/ompeval/HandEvaluator.cpp)
target_include_directories(poker_core PUBLIC cpp)
target_link_libraries(poker_core PUBLIC Threads::Threads)
add_executable(poker_generate_tables tools/generate_tables.cpp)
target_link_libraries(poker_generate_tables PRIVATE poker_core)
add_executable(poker_evaluator_benchmark benchmarks/evaluator.cpp)
target_link_libraries(poker_evaluator_benchmark PRIVATE poker_core)
enable_testing()
add_executable(poker_exhaustive tests/exhaustive.cpp)
target_link_libraries(poker_exhaustive PRIVATE poker_core)
add_test(NAME exhaustive COMMAND poker_exhaustive)
add_executable(poker_stress tests/stress.cpp)
target_link_libraries(poker_stress PRIVATE poker_core)
add_test(NAME stress COMMAND poker_stress)
